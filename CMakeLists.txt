cmake_minimum_required(VERSION 3.20)
project(aether5g VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(AETHER_SANITIZER "" CACHE STRING "address, undefined, or thread")
if(AETHER_SANITIZER)
  add_compile_options(-fsanitize=${AETHER_SANITIZER} -fno-omit-frame-pointer -fno-sanitize-recover=all)
  add_link_options(-fsanitize=${AETHER_SANITIZER})
endif()
add_library(aether_core src/scheduler.cpp)
target_include_directories(aether_core PUBLIC include)
target_compile_options(aether_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(aether_sim src/main.cpp)
target_link_libraries(aether_sim PRIVATE aether_core)
enable_testing()
add_executable(aether_tests tests/scheduler_test.cpp)
target_link_libraries(aether_tests PRIVATE aether_core)
add_test(NAME scheduler_invariants COMMAND aether_tests)

set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.15.2.tar.gz URL_HASH SHA256=7b42b4d6ed48810c5362c265a17faebe90dc2373c885e5216439d37927f02926 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(googletest)
add_library(aether_mac src/mac/scheduler.cpp)
target_include_directories(aether_mac PUBLIC include)
target_compile_options(aether_mac PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(mac_tests tests/mac_test.cpp)
target_link_libraries(mac_tests PRIVATE aether_mac aether_core GTest::gtest_main)
include(GoogleTest)
gtest_discover_tests(mac_tests)
add_executable(aether_slots simulator/main.cpp)
target_link_libraries(aether_slots PRIVATE aether_mac)
add_executable(scheduler_bench benchmarks/scheduler_bench.cpp)
target_link_libraries(scheduler_bench PRIVATE aether_mac)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME scenario_replay COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/scenario_test.py $<TARGET_FILE:aether_slots>)

add_library(aether_sprint2 src/mac/harq.cpp src/mac/controller.cpp src/phy/feedback.cpp src/mac/bidirectional.cpp)
target_include_directories(aether_sprint2 PUBLIC include)
target_link_libraries(aether_sprint2 PUBLIC aether_mac)
target_compile_options(aether_sprint2 PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(sprint2_tests tests/harq_test.cpp tests/controller_test.cpp tests/phy_test.cpp tests/bidirectional_test.cpp)
target_link_libraries(sprint2_tests PRIVATE aether_sprint2 GTest::gtest_main)
gtest_discover_tests(sprint2_tests)

add_executable(aether_bidirectional simulator/bidirectional.cpp)
target_link_libraries(aether_bidirectional PRIVATE aether_sprint2)
target_compile_options(aether_bidirectional PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME bidirectional_cli_replay COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/bidirectional_cli_test.py $<TARGET_FILE:aether_bidirectional>)
set_tests_properties(bidirectional_cli_replay PROPERTIES TIMEOUT 120)
